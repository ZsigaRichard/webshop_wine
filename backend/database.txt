DROP DATABASE wine_webshop;
CREATE DATABASE wine_webshop;

CREATE TABLE Users (
    userID INT NOT NULL AUTO_INCREMENT,
    email VARCHAR(320) NOT NULL,
    password VARCHAR(255) NOT NULL,
    firstname VARCHAR(255) NOT NULL,
    lastname VARCHAR(255) NOT NULL,
    lastlogin DATETIME,
    PRIMARY KEY (userID)
);

CREATE TABLE Shipping_address(
    shipping_addressID INT NOT NULL AUTO_INCREMENT,
    userID INT NOT NULL,
    name VARCHAR(512) NOT NULL,
    phone_number VARCHAR(16) NOT NULL,
    postcode  VARCHAR(10) NOT NULL,
    address VARCHAR(255) NOT NULL,
    floor VARCHAR(3) NOT NULL,
    city VARCHAR(128) NOT NULL,
    door VARCHAR(128),

    PRIMARY KEY (shipping_addressID),
    FOREIGN KEY (userID) REFERENCES Users(userID)
);

CREATE TABLE Billing_address(
    billing_addressID INT NOT NULL AUTO_INCREMENT,
    userID INT,
    name VARCHAR(512) NOT NULL,
    phone_number VARCHAR(16) NOT NULL,
    postcode  VARCHAR(10) NOT NULL,
    address VARCHAR(255) NOT NULL,
    floor VARCHAR(3) NOT NULL,
    city VARCHAR(128) NOT NULL,
    door VARCHAR(128),
    email VARCHAR(320) NOT NULL,

    PRIMARY KEY (billing_addressID),
    FOREIGN KEY (userID) REFERENCES Users(userID)
);

CREATE TABLE Orders(
    orderID INT NOT NULL AUTO_INCREMENT,
    status ENUM('Feldolgozás alatt','Csomagolás','Feladva'),
    paying_method ENUM('utánvét', 'bankkártya', 'PayPal') NOT NULL,
    userID int NOT NULL,
    Name VARCHAR(512),
    billing_addressID INT NOT NULL,
    shipping_addressID INT NOT NULL,
    tracking_number VARCHAR(64),
    total INT NOT NULL,
    date DATETIME,

    PRIMARY KEY (orderID),
    FOREIGN KEY (userID) REFERENCES Users(userID),
    FOREIGN KEY (billing_addressID) REFERENCES Billing_address(billing_addressID),
    FOREIGN KEY (shipping_addressID) REFERENCES Shipping_address(shipping_addressID)
);

CREATE TABLE Products(
    product_name varchar(255) NOT NULL,
    productID int NOT NULL AUTO_INCREMENT,
    vintage YEAR,
    alcohol_content TINYINT,
    size DECIMAL(4,2),
    country varchar(64),
    wine_region varchar(64),
    sweetness varchar(16),
    type varchar(16),
    color varchar(16),
    variety varchar(255),
    image varchar(255),
    description text,
    stock SMALLINT,
    price MEDIUMINT,
    
    PRIMARY KEY (productID)

);

CREATE TABLE Ordered_items(
    order_itemsID INT NOT NULL AUTO_INCREMENT,
    orderID INT,
    productID INT,
    amount SMALLINT,

    PRIMARY KEY (order_itemsID),
    FOREIGN KEY (orderID) REFERENCES Orders(orderID),
    FOREIGN KEY (productID) REFERENCES Products(productID)
);
